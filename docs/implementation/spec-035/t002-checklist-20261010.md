"""035 T002 checklist (research.md sections A and B), applied mechanically. Exit 0 = pass."""
from datetime import datetime
import json, re, sys
from pathlib import Path

EV = Path("docs/implementation/spec-035")
KEYS = ("nominal_or_adjusted_ohlc", "split_dividend_fields", "corporate_action_history_depth",
        "dividend_amount_basis", "coverage_start")
SECRET = re.compile(r"key|token|secret|auth|password", re.I)
# spec §3 rows in order: (candidate word in the role column, T001 record the row may cite or None)
ROWS = (("Tiingo", "tiingo"), ("Tiingo", "tiingo"), ("Alpaca", "alpaca"), ("Alpaca", "alpaca"),
        ("EODHD", "eodhd"), ("yfinance", None), ("SEC EDGAR", None))
ESTABLISHED = {6: ".specify/specs/044-nominal-price-reconstruction/spec.md"}  # the only non-T001 cite T002 names
HEDGES = ("not established", "not observed", "UNVERIFIED", "not stated")


def utc(value) -> bool:
    try:
        return datetime.fromisoformat(str(value)).utcoffset().total_seconds() == 0
    except (ValueError, AttributeError):
        return False


def record(path: Path) -> dict:
    return json.loads(re.search(r"```json\n(.*?)\n```", path.read_text(encoding="utf-8"), re.S).group(1))


def check_record(r: dict) -> list[str]:
    bad = []
    if not r.get("source") or not utc(r.get("fetched_at")):
        bad.append("A1 source/fetched_at UTC")
    if not {("AAPL", "2020-08-31"), ("NVDA", "2024-06-10")} <= {(w.get("symbol"), w.get("split")) for w in r.get("windows", [])}:
        bad.append("A2 windows")
    for q in r.get("requests", []):
        text = [str(q.get("endpoint_path", ""))] + [f"{k}={v}" for k, v in dict(q.get("parameters") or {}).items()]
        if not q.get("endpoint_path") or q.get("parameters") is None or not utc(q.get("requested_at_utc")) \
                or any(SECRET.search(t) for t in text):
            bad.append(f"A3 request {q.get('endpoint_path')}")
    obs = r.get("observations", {})
    for k in KEYS:
        o = obs.get(k)
        if not o or not o.get("state") or (o["state"] == "UNVERIFIED" and not str(o.get("reason", "")).strip()):
            bad.append(f"A4 {k}")
    return bad


def check_rows(text: str, root: Path) -> list[str]:
    rows = [[c.strip() for c in l.strip().strip("|").split("|")] for l in text.splitlines()
            if re.match(r"\|\s*\d+\s*\|", l)]
    bad = [] if len(rows) == len(ROWS) else [f"B1 {len(rows)} rows, expected {len(ROWS)}"]
    for (n, role, claim, obs, state, date, ev), (word, source) in zip(rows, ROWS):
        n = int(n)
        if word not in role:
            bad.append(f"B1 row {n}: role {role!r} is not spec §3's {word} row")
        if state not in ("OBSERVED", "PARTIAL", "ESTABLISHED", "UNVERIFIED"):
            bad.append(f"B2 row {n}: unknown state {state}")
        if state == "UNVERIFIED":
            bad += [] if len(ev.removeprefix("Reason:").strip()) > 10 and ev.startswith("Reason:") else [f"B3 row {n}: no reason"]
            continue
        cites = [c.split("#", 1) for c in re.findall(r"`([^`]+#[^`]+)`", ev)]
        if state == "ESTABLISHED":
            if not cites or any(path != ESTABLISHED.get(n) for path, _ in cites):
                bad.append(f"B2 row {n}: ESTABLISHED needs the non-T001 evidence T002 names")
        elif not cites or any(path != f"fr001-{source}-20261009.md" for path, _ in cites):
            bad.append(f"B2 row {n}: {state} must cite only this row's own T001 record")
        if state == "OBSERVED" and any(h in obs for h in HEDGES):
            bad.append(f"B2 row {n}: OBSERVED but the observation leaves a part unestablished")
        for path, key in cites:
            p = Path(path) if "/" in path else EV / path
            if not (root / p).is_file():
                bad.append(f"B2 row {n}: {p} missing"); continue
            if p.name.startswith("fr001-"):
                r = record(root / p)
                if r["observations"].get(key, {}).get("state", "UNVERIFIED") == "UNVERIFIED":
                    bad.append(f"B2 row {n}: {key} unobserved in {p.name}")
                if str(r.get("fetched_at", ""))[:10] != date:
                    bad.append(f"B2 row {n}: date {date} != record fetched_at {r.get('fetched_at')}")
    return bad


if __name__ == "__main__":
    root = Path(sys.argv[1])
    bad = [f"{p.name}: {b}" for p in sorted((root / EV).glob("fr001-*.md")) for b in check_record(record(p))]
    bad += check_rows((root / sys.argv[2]).read_text(encoding="utf-8"), root)
    print("\n".join(bad) or "PASS"); sys.exit(1 if bad else 0)
